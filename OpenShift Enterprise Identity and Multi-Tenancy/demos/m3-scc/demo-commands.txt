oc get scc -o name
clear
oc describe scc restricted-v2  # confirm
oc describe scc privileged  # confirm

# Reproduce failure (default SA)
oc apply -f privileged-pod-default.yaml
oc get pod privileged-demo
oc get events --sort-by=.lastTimestamp | tail -n 30

# Copy + split the event message by comma to make it readable
EVT_MSG="$(oc get events --field-selector involvedObject.kind=ReplicaSet --sort-by=.lastTimestamp -o jsonpath='{.items[-1:].message}')"
printf '%s\n' "${EVT_MSG}"
printf '%s\n' "${EVT_MSG}" | tr ',' '\n'

# Confirm which SA the pod is using (empty means "default" in many cases)
oc get deployment privileged-demo -o jsonpath='{.spec.template.spec.serviceAccountName}{"\n"}'
oc auth can-i use scc/privileged --as="system:serviceaccount:scc-demo:default"
oc adm policy who-can use scc privileged

# Fix (dedicated SA + RBAC grant to SCC)
oc create sa privileged-app -n scc-demo
oc adm policy add-scc-to-user privileged -z privileged-app -n scc-demo
oc auth can-i use scc/privileged --as="system:serviceaccount:scc-demo:privileged-app"

# Show the RBAC object that was created (RoleBinding vs ClusterRoleBinding varies by release/usage)
oc describe rolebinding system:openshift:scc:privileged

# Redeploy pod with explicit serviceAccountName
oc apply -f privileged-pod-with-sa.yaml
oc get pod

POD_NAME="$(oc get pods -l app=privileged-demo-with-sa -o jsonpath='{.items[0].metadata.name}')"
oc get pod "${POD_NAME}" -o jsonpath='{.metadata.annotations.openshift\.io/scc}{"\n"}'
