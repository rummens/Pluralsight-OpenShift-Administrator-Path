# BASELINE
oc get nodes -o wide #wait 5
oc get pods -n drain-demo -o wide #wait 5

# BREAK NODE — STOP AND MASK KUBELET chroot /host bash -c "systemctl stop kubelet"
WORKER=$(oc get nodes -l node-role.kubernetes.io/worker= -o jsonpath='{.items[0].metadata.name}')
oc debug node/$WORKER  #confirm

# WATCH NODE GO NOTREADY
watch oc get nodes #confirm

# INVESTIGATE NODE CONDITIONS
oc describe node $WORKER

# CORDON AND DRAIN
oc adm cordon $WORKER
oc get pods -n drain-demo -o wide #wait 5

# RECOVER VIA SSH (KUBELET RESTART)
ssh -i $HOME/.ssh/id_ocp core@$WORKER_IP "sudo systemctl start kubelet"

# WATCH RECOVERY
watch oc get nodes #confirm
oc adm uncordon $WORKER
oc get nodes #wait 5

# POST-RECOVERY VALIDATION
oc get pvc -A | grep -v Bound #wait 3
oc get pods -A | grep -v Running | grep -v Completed #wait 5
oc get mcp #wait 5
