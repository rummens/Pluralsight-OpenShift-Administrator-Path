# BROAD VIEW
oc adm top nodes
oc adm top pods -n globomantics

# NODE DEEP-DIVE
WORKER=$(oc get nodes -l node-role.kubernetes.io/worker -o jsonpath='{.items[1].metadata.name}')
echo $WORKER
oc describe node $WORKER | grep -A 5 "Allocated resources" #wait 8

# Find bad pods with high resource requests
oc get pods -A -o json | jq -r '
  .items[]
  | select(.metadata.namespace | startswith("openshift-") | not)
  | select(any(
      .spec.containers[];
      (.resources.requests.cpu // "0") as $c
      | (if ($c | endswith("m"))
         then ($c | rtrimstr("m") | tonumber) / 1000
         else ($c | tonumber) end) > 6
    ))
  | "\(.metadata.namespace)/\(.metadata.name)"'


# PROMQL IN CONSOLE
echo "" #confirm

# EVENTS AND OOMKILLED POD
oc get events -n globomantics --sort-by='.lastTimestamp' | tail -20

# PENDING POD DIAGNOSIS
oc get pod pending-test -n globomantics
oc describe pod pending-test -n globomantics | grep -A 10 "Events"
oc delete pod pending-test -n globomantics

# FINAL STATE
oc get pods -n globomantics -l app=web
